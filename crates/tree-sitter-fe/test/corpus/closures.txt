fn closure_modes<T: Trait>() {
    let never = |value: view !| value
    let own_mode = |
        value: own
    | value
    let ref_mode = |
        value: ref
    | value
    let mut_mode = |
        value: mut
    | value
    let view_mode = |
        value: view
    | value
}

==================
Contextual view types with trivia (Tree-sitter editor recovery)
==================

fn contextual_view<T: Trait>() {
    let line_comment =
        |value: view <T // > inside trivia
            as Trait>::Item| value
    let line_comments_at_boundaries =
        |value: view // before the qualified type
            <T // inside the qualified type
            as Trait> // before the associated item separator
            ::Item| value
    let block_comment =
        |value: view <T /* editor recovery only: > */ as Trait>::Item| value
    let nested_block_comment =
        |value: view <T /* outer /* nested > */ end */ as Trait>::Item| value
    let block_comments_at_boundaries =
        |value: view /* before */ <T /* inside */ as Trait> /* after */ ::Item| value
    let string =
        |value: view <Holder<" > as "> as Trait>::Item| value
    let const_comparison =
        |value: view <Holder<{ 2 > 1 }> as Trait>::Item| value
    let parenthesized_const_comparison =
        |value: view <Holder<{ (2 > 1) }> as Trait>::Item| value
    let bracketed_const_comparison =
        |value: view <Holder<{ [2 > 1] }> as Trait>::Item| value
    let generic = |value: view<T>| value
    let qualified = |value: view::Nested| value
}
